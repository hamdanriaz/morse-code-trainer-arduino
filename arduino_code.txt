  digitalWrite(inputSpace, HIGH);
  delay(do_space);
  digitalWrite(inputSpace, LOW);

}

void dash() {
  digitalWrite(inputSpace, HIGH);
  delay(da_space);
  digitalWrite(inputSpace, LOW);
  delay(low_space);
}

String random_string(){
  String x = "";
  int i = 0;

  int alphaNumerals = random(3, 11);

  while (i < alphaNumerals) {
    int alpha_or_numeral = random(0, 2);

    if (alpha_or_numeral == 0) {
      char randomLetter = (char)random(65, 91);
      x += randomLetter;
      i += 1;
    } else {
      int randomNumber = random(0, 10);
      x += randomNumber;
      i += 1;
    }
  }

  return x;
}

void setup() {
  pinMode(inputSpace, OUTPUT);
  Serial.begin(9600);

  Serial.println("--- Morse Practice Game ---");
  Serial.println("Press Enter in the Serial Monitor to start...");
  
  while (!Serial.available()) {}
  randomSeed(micros());
  Serial.readString();
}

void loop() {
  String randStr = random_string();
  randStr.toUpperCase();

  Serial.println("\nFlashing new code...");

  for (int i = 0; i < randStr.length(); i++) {
    char c = randStr[i];
    if (c == 'A') {morse_a(); delay(L_space);}
